docker run -d \

 -p 9042:9000 \

 -p 9043:9001 \

 --name minio1 \

 -v /home/minio/data:/data \

 -e "MINIO_ROOT_USER=AKIAIOSFODNN7EXAMPLE" \

 -e "MINIO_ROOT_PASSWORD=wJalrXUtnFEMI/K7MDENG/bPxRfiCYEXAMPLEKEY" \

 minio/minio:latest server /data --console-address ":9001"
